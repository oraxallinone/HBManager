    // App State
    let tasks = [];
    let expandedSubtasks = new Set(); // tracks open accordion task IDs

    const emptyStateEl = document.getElementById('emptyState');
    const totalBadgeEl = document.getElementById('totalBadge');
    const pendingBadgeEl = document.getElementById('pendingBadge');
    const completedBadgeEl = document.getElementById('completedBadge');
    const searchInput = document.getElementById('searchInput');
    const clearSearchBtn = document.getElementById('clearSearchBtn');

    const statusFilter = document.getElementById('statusFilter');

    // Modal Elements
    const taskModal = document.getElementById('taskModal');
    const modalOverlay = document.getElementById('modalOverlay');
    const taskForm = document.getElementById('taskForm');
    const modalTitle = document.getElementById('modalTitle');
    const taskIdInput = document.getElementById('taskId');
    const taskTitleInput = document.getElementById('taskTitleInput');

    const taskDueDateInput = document.getElementById('taskDueDateInput');

    // Toast Notification helper
    function showToast(message) {
      const toast = document.getElementById('toast');
      const toastMsg = document.getElementById('toastMsg');
      toastMsg.textContent = message;
      toast.classList.remove('translate-y-20', 'opacity-0');
      toast.classList.add('translate-y-0', 'opacity-100');

        toast.classList.remove('translate-y-0', 'opacity-100');
        toast.classList.add('translate-y-20', 'opacity-0');
      }, 2600);
    }

    // Initialize State from LocalStorage
    function loadTasks() {
      const stored = localStorage.getItem('taskmaster_data_v1');
      if (stored) {
        try {
          tasks = JSON.parse(stored);
        } catch (e) {
          tasks = DEFAULT_TASKS;
        }
      } else {
        tasks = DEFAULT_TASKS;
        saveTasks();
      }
      renderTasks();
    }

    // Save State to LocalStorage
    function saveTasks() {
      localStorage.setItem('taskmaster_data_v1', JSON.stringify(tasks));
      renderCounters();
    }

    // Calculate & Render Counters
    function renderCounters() {
      const total = tasks.length;
      const completed = tasks.filter(t => t.completed).length;
      const pending = total - completed;

      totalBadgeEl.textContent = total;
      completedBadgeEl.textContent = completed;
      pendingBadgeEl.textContent = pending;
    }

    // Format Date for Card UI
    function formatDateDisplay(dateStr) {
      if (!dateStr) return 'No due date';
      const parts = dateStr.split('-');
      if (parts.length !== 3) return dateStr;
      const d = new Date(parts[0], parts[1] - 1, parts[2]);
      return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
    }

    // Priority badge styling helper
    function getPriorityBadge(priority) {
      switch (priority) {
        case 'High':
          return '<span class="inline-flex items-center px-2 py-0.5 rounded-full text-2xs font-semibold bg-rose-50 text-rose-700 border border-rose-200">High</span>';
        case 'Low':
          return '<span class="inline-flex items-center px-2 py-0.5 rounded-full text-2xs font-semibold bg-slate-100 text-slate-600 border border-slate-200">Low</span>';
        default:
          return '<span class="inline-flex items-center px-2 py-0.5 rounded-full text-2xs font-semibold bg-amber-50 text-amber-700 border border-amber-200">Medium</span>';
      }
    }

    // Escape HTML to prevent XSS
    function escapeHtml(str) {
      if (!str) return '';
      return String(str)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#039;');
    }

    // Main Render Routine
    function renderTasks() {
      const searchTerm = searchInput.value.trim().toLowerCase();
      const priorityVal = priorityFilter.value;
      const statusVal = statusFilter.value;

      // Filter tasks
      const filteredTasks = tasks.filter(task => {
        // Priority filter
        if (priorityVal !== 'ALL' && task.priority !== priorityVal) return false;

        // Status filter
        if (statusVal === 'PENDING' && task.completed) return false;
        if (statusVal === 'COMPLETED' && !task.completed) return false;

        // Search text filter across title, description, and subtasks
        if (searchTerm) {
          const matchTitle = task.title.toLowerCase().includes(searchTerm);
          const matchDesc = (task.description || '').toLowerCase().includes(searchTerm);
          const matchSubtask = (task.subtasks || []).some(sub => sub.title.toLowerCase().includes(searchTerm));
          if (!matchTitle && !matchDesc && !matchSubtask) return false;
        }

        return true;
      });

      renderCounters();

      // Empty State Handling
      if (filteredTasks.length === 0) {
        taskListEl.innerHTML = '';
        emptyStateEl.classList.remove('hidden');
        emptyStateEl.classList.add('flex');
        return;
      } else {
        emptyStateEl.classList.add('hidden');
        emptyStateEl.classList.remove('flex');
      }

      // Render cards
      taskListEl.innerHTML = filteredTasks.map(task => {
        const subtasks = task.subtasks || [];
        const completedSubs = subtasks.filter(s => s.completed).length;
        const totalSubs = subtasks.length;
        const isAccordionOpen = expandedSubtasks.has(task.id);

        const cardCompletedClass = task.completed ? 'bg-slate-50/90 border-slate-200 opacity-90' : 'bg-white border-slate-200 shadow-sm hover:shadow-md';
        const titleClass = task.completed ? 'line-through text-slate-400' : 'text-slate-800 font-semibold';

        return `
          <div class="group relative rounded-2xl border p-5 transition-all duration-200 flex flex-col justify-between ${cardCompletedClass}" data-task-id="${task.id}">
            <div>
              <!-- Card Header: Status Toggle, Title, Priority -->
              <div class="flex items-start justify-between gap-3">
                <div class="flex items-start gap-3 flex-1 min-w-0">
                  <button 
                    onclick="toggleTaskStatus('${task.id}')"
                    class="mt-0.5 w-5 h-5 rounded-lg border flex items-center justify-center transition-colors ${
                      task.completed 
                        ? 'bg-emerald-500 border-emerald-500 text-white' 
                        : 'border-slate-300 hover:border-indigo-500 bg-white'
                    }"
                    title="${task.completed ? 'Mark pending' : 'Mark completed'}"
                  >
                    ${task.completed ? `
                      <svg class="w-3.5 h-3.5" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                        <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2.5" d="M5 13l4 4L19 7"></path>
                      </svg>
                    ` : ''}
                  </button>
                  <div class="flex-1 min-w-0">
                    <h3 class="text-base truncate leading-snug ${titleClass}" title="${escapeHtml(task.title)}">
                      ${escapeHtml(task.title)}
                    </h3>
                  </div>
                </div>
                <div>${getPriorityBadge(task.priority)}</div>
              </div>

              <!-- Description -->
              ${task.description ? `
                <p class="mt-2.5 text-xs text-slate-500 line-clamp-2 leading-relaxed">
                  ${escapeHtml(task.description)}
                </p>
              ` : ''}

              <!-- Due Date Indicator -->
              <div class="mt-4 flex items-center gap-1.5 text-xs font-medium text-slate-500">
                <svg class="w-3.5 h-3.5 text-slate-400" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                  <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M8 7V3m8 4V3m-9 8h10M5 21h14a2 2 0 002-2V7a2 2 0 00-2-2H5a2 2 0 00-2 2v12a2 2 0 002 2z"></path>
                </svg>
                <span>${formatDateDisplay(task.dueDate)}</span>
              </div>
            </div>

            <!-- Card Bottom: Subtasks Accordion & Action Buttons -->
            <div class="mt-4 pt-3 border-t border-slate-100 flex flex-col gap-2">
              
              <!-- Accordion Toggle Button -->
              <div class="flex items-center justify-between">
                <button 
                  onclick="toggleSubtaskAccordion('${task.id}')"
                  class="text-xs font-semibold text-slate-600 hover:text-indigo-600 flex items-center gap-1.5 py-1 transition-colors"
                >
                  <span class="text-slate-400">
                    <svg class="w-3.5 h-3.5 transform transition-transform ${isAccordionOpen ? 'rotate-180' : ''}" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                      <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M19 9l-7 7-7-7"></path>
                    </svg>
                  </span>
                  <span>Subtasks (${completedSubs}/${totalSubs})</span>
                </button>

                <!-- Actions: Edit & Delete -->
                <div class="flex items-center gap-1">
                  <button 
                    onclick="editTask('${task.id}')"
                    class="p-1.5 rounded-lg text-slate-400 hover:text-indigo-600 hover:bg-indigo-50 transition"
                    title="Edit Task"
                  >
                    <svg class="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                      <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M15.232 5.232l3.536 3.536m-2.036-5.036a2.5 2.5 0 113.536 3.536L6.5 21.036H3v-3.572L16.732 3.732z"></path>
                    </svg>
                  </button>
                  <button 
                    onclick="deleteTask('${task.id}')"
                    class="p-1.5 rounded-lg text-slate-400 hover:text-rose-600 hover:bg-rose-50 transition"
                    title="Delete Task"
                  >
                    <svg class="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                      <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M19 7l-.867 12.142A2 2 0 0116.138 21H7.862a2 2 0 01-1.995-1.858L5 7m5 4v6m4-6v6m1-10V4a1 1 0 00-1-1h-4a1 1 0 00-1 1v3M4 7h16"></path>
                    </svg>
                  </button>
                </div>
              </div>

              <!-- Collapsible Subtasks Container -->
              ${isAccordionOpen ? `
                <div class="mt-2 space-y-1.5 bg-slate-50/80 p-2.5 rounded-xl border border-slate-100 transition-all">
                  ${totalSubs === 0 ? `
                    <p class="text-2xs text-slate-400 italic py-1">No subtasks assigned.</p>
                  ` : subtasks.map(sub => `
                    <div class="flex items-center justify-between text-xs py-1 px-1.5 rounded-lg hover:bg-white transition-colors group/sub">
                      <label class="flex items-center gap-2 cursor-pointer flex-1 min-w-0 pr-2">
                        <input 
                          type="checkbox" 
                          ${sub.completed ? 'checked' : ''} 
                          onchange="toggleSubtaskStatus('${task.id}', '${sub.id}')"
                          class="w-3.5 h-3.5 rounded text-indigo-600 focus:ring-indigo-500 border-slate-300 transition"
                        />
                        <span class="truncate ${sub.completed ? 'line-through text-slate-400' : 'text-slate-700 font-medium'}">
                          ${escapeHtml(sub.title)}
                        </span>
                      </label>
                      <span class="text-2xs text-slate-400 whitespace-nowrap pl-1">
                        ${formatDateDisplay(sub.dueDate)}
                      </span>
                    </div>
                  `).join('')}
                </div>
              ` : ''}

            </div>
          </div>
        `;
      }).join('');
    }

    // Accordion Toggle
    window.toggleSubtaskAccordion = function(taskId) {
      if (expandedSubtasks.has(taskId)) {
        expandedSubtasks.delete(taskId);
      } else {
        expandedSubtasks.add(taskId);
      }
      renderTasks();
    };

    // Toggle Task Complete State
    window.toggleTaskStatus = function(taskId) {
      const task = tasks.find(t => t.id === taskId);
      if (task) {
        task.completed = !task.completed;
        saveTasks();
        renderTasks();
        showToast(task.completed ? 'Task marked complete' : 'Task marked pending');
      }
    };

    // Toggle Nested Subtask State
    window.toggleSubtaskStatus = function(taskId, subtaskId) {
      const task = tasks.find(t => t.id === taskId);
      if (task && task.subtasks) {
        const sub = task.subtasks.find(s => s.id === subtaskId);
        if (sub) {
          sub.completed = !sub.completed;
          // Check if all subtasks complete, optionally update task
          saveTasks();
          renderTasks();
        }
      }
    };

    // Delete Task
    window.deleteTask = function(taskId) {
      tasks = tasks.filter(t => t.id !== taskId);
      expandedSubtasks.delete(taskId);
      saveTasks();
      renderTasks();
      showToast('Task deleted');
    };

    // Modal Operations
    function openModal(isEdit = false, taskData = null) {
      modalSubtasksContainer.innerHTML = '';
      if (isEdit && taskData) {
        modalTitle.textContent = 'Edit Task';
        taskIdInput.value = taskData.id;
        taskTitleInput.value = taskData.title;
        taskDescInput.value = taskData.description || '';
        taskDueDateInput.value = taskData.dueDate || '';
        taskPriorityInput.value = taskData.priority || 'Medium';

        // Render subtasks into modal
        (taskData.subtasks || []).forEach(sub => {
          addSubtaskRow(sub.id, sub.title, sub.dueDate, sub.completed);
        });
      } else {
        modalTitle.textContent = 'Add New Task';
        taskForm.reset();
        taskIdInput.value = '';
        // Set default date to today
        const today = new Date().toISOString().split('T')[0];
        taskDueDateInput.value = today;
        taskPriorityInput.value = 'Medium';
      }
      updateSubtaskEmptyState();
      taskModal.classList.remove('hidden');
      taskTitleInput.focus();
    }

    function closeModal() {
      taskModal.classList.add('hidden');
      taskForm.reset();
      modalSubtasksContainer.innerHTML = '';
    }

    // Modal Subtasks Dynamic Row Addition
    function addSubtaskRow(id = '', title = '', dueDate = '', completed = false) {
      const subId = id || 'sub-' + Date.now() + '-' + Math.random().toString(36).substr(2, 4);
      const row = document.createElement('div');
      row.className = 'subtask-row flex items-center gap-2 bg-white p-2 rounded-lg border border-slate-200';
      row.dataset.subId = subId;
      row.dataset.completed = completed ? 'true' : 'false';

      row.innerHTML = `
        <input 
          type="text" 
          placeholder="Subtask title" 
          value="${escapeHtml(title)}"
          required
          class="subtask-title-input flex-1 px-2.5 py-1 text-xs bg-slate-50 border border-slate-200 rounded-lg focus:bg-white focus:outline-none focus:border-indigo-500"
        />
        <input 
          type="date" 
          value="${dueDate || ''}"
          required
          class="subtask-date-input w-36 px-2 py-1 text-xs bg-slate-50 border border-slate-200 rounded-lg focus:bg-white focus:outline-none focus:border-indigo-500"
        />
        <button 
          type="button" 
          class="remove-subtask-btn p-1 text-slate-400 hover:text-rose-600 rounded-lg transition"
          title="Remove subtask"
        >
          <svg class="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24">
            <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M6 18L18 6M6 6l12 12"></path>
          </svg>
        </button>
      `;

      row.querySelector('.remove-subtask-btn').addEventListener('click', () => {
        row.remove();
        updateSubtaskEmptyState();
      });

      modalSubtasksContainer.appendChild(row);
      updateSubtaskEmptyState();
    }

    function updateSubtaskEmptyState() {
      if (modalSubtasksContainer.children.length === 0) {
        subtasksEmptyNote.classList.remove('hidden');
      } else {
        subtasksEmptyNote.classList.add('hidden');
      }
    }

    // Edit Task Trigger
    window.editTask = function(taskId) {
      const task = tasks.find(t => t.id === taskId);
      if (task) {
        openModal(true, task);
      }
    };

    // Modal Form Submission (Create or Update)
    taskForm.addEventListener('submit', (e) => {
      e.preventDefault();

      const id = taskIdInput.value;
      const title = taskTitleInput.value.trim();
      const description = taskDescInput.value.trim();
      const dueDate = taskDueDateInput.value;
      const priority = taskPriorityInput.value;

      if (!title || !dueDate) return;

      // Extract subtasks from modal DOM rows
      const subtaskRows = modalSubtasksContainer.querySelectorAll('.subtask-row');
      const subtasks = Array.from(subtaskRows).map(row => {
        const subId = row.dataset.subId;
        const subCompleted = row.dataset.completed === 'true';
        const subTitle = row.querySelector('.subtask-title-input').value.trim();
        const subDate = row.querySelector('.subtask-date-input').value;
        return {
          id: subId,
          title: subTitle,
          dueDate: subDate,
          completed: subCompleted
        };
      }).filter(s => s.title !== '');

      if (id) {
        // Update existing task
        const taskIdx = tasks.findIndex(t => t.id === id);
        if (taskIdx !== -1) {
          tasks[taskIdx] = {
            ...tasks[taskIdx],
            title,
            description,
            dueDate,
            priority,
            subtasks
          };
          showToast('Task updated successfully');
        }
      } else {
        // Create new task
        const newTask = {
          id: 'task-' + Date.now(),
          title,
          description,
          dueDate,
          priority,
          completed: false,
          subtasks
        };
        tasks.unshift(newTask);
        showToast('New task added');
      }

      saveTasks();
      closeModal();
      renderTasks();
    });

    // Search & Filter Listeners
    searchInput.addEventListener('input', () => {
      if (searchInput.value.trim().length > 0) {
        clearSearchBtn.classList.remove('hidden');
      } else {
        clearSearchBtn.classList.add('hidden');
      }
      renderTasks();
    });

    clearSearchBtn.addEventListener('click', () => {
      searchInput.value = '';
      clearSearchBtn.classList.add('hidden');
      renderTasks();
      searchInput.focus();
    });

    priorityFilter.addEventListener('change', renderTasks);
    statusFilter.addEventListener('change', renderTasks);

    // Modal Trigger Listeners
    openModalBtn.addEventListener('click', () => openModal(false));
    openModalBtnMobile.addEventListener('click', () => openModal(false));
    emptyActionBtn.addEventListener('click', () => openModal(false));
    closeModalBtn.addEventListener('click', closeModal);
    cancelModalBtn.addEventListener('click', closeModal);
    modalOverlay.addEventListener('click', closeModal);

    addSubtaskBtn.addEventListener('click', () => {
      const today = new Date().toISOString().split('T')[0];
      addSubtaskRow('', '', today, false);
    });

    // Close modal on ESC key
    document.addEventListener('keydown', (e) => {
      if (e.key === 'Escape' && !taskModal.classList.contains('hidden')) {
        closeModal();
      }
    });

    // Export Data to JSON File
    function exportToJSON() {
      const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(tasks, null, 2));
      const downloadAnchor = document.createElement('a');
      downloadAnchor.setAttribute("href", dataStr);
      downloadAnchor.setAttribute("download", `taskmaster_export_${new Date().toISOString().split('T')[0]}.json`);
      document.body.appendChild(downloadAnchor);
      downloadAnchor.click();
      downloadAnchor.remove();
      showToast('Exported tasks to JSON');
    }

    exportTasksBtn.addEventListener('click', exportToJSON);
    exportTasksMobileBtn.addEventListener('click', exportToJSON);

    // Initial Execution
    window.addEventListener('DOMContentLoaded', () => {
      loadTasks();
    });
  </script>
</body>

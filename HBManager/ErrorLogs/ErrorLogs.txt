<script>
    let lastAddedValues = null;
    let editingRow = null;
    let rowToDelete = null;

    function initDropdowns() {
        const now = new Date();
        const currentMonth = now.getMonth();
        const currentYear = now.getFullYear();

        document.getElementById('monthSelect').value = currentMonth;

        const yearSelect = document.getElementById('yearSelect');
        for (let y = currentYear - 5; y <= currentYear + 5; y++) {
            const opt = document.createElement('option');
            opt.value = y;
            opt.textContent = y;
            if (y === currentYear) opt.selected = true;
            yearSelect.appendChild(opt);
        }
    }

    function setCurrentDateTime() {
        const now = new Date();
        const year = now.getFullYear();
        const month = String(now.getMonth() + 1).padStart(2, '0');
        const day = String(now.getDate()).padStart(2, '0');
        const hours = String(now.getHours()).padStart(2, '0');
        const minutes = String(now.getMinutes()).padStart(2, '0');

        const localDateTime = `${year}-${month}-${day}T${hours}:${minutes}`;
        document.getElementById('entryDateTime').value = localDateTime;
    }

    window.addEventListener('DOMContentLoaded', () => {
        initDropdowns();
        setCurrentDateTime();
    });

    // Keyboard shortcuts: Tab trap in entry modal + Escape key support for modals
    document.addEventListener('keydown', function(e) {
        const entryModal = document.getElementById('entryModal');
        const deleteModal = document.getElementById('deleteConfirmModal');

        if (e.key === 'Escape') {
            if (deleteModal.style.display === 'flex') closeDeleteModal();
            else if (entryModal.style.display === 'flex') closeModal();
        }

        if (entryModal.style.display === 'flex' && e.key === 'Tab') {
            e.preventDefault();
            const inputs = Array.from(entryModal.querySelectorAll('.val-input'));
            const currentIndex = inputs.indexOf(document.activeElement);

            let nextIndex;
            if (currentIndex === -1) {
                nextIndex = 0;
            } else if (e.shiftKey) {
                nextIndex = (currentIndex - 1 + inputs.length) % inputs.length;
            } else {
                nextIndex = (currentIndex + 1) % inputs.length;
            }

            inputs[nextIndex].focus();
            inputs[nextIndex].select();
        }
    });

    function openModal() {
        editingRow = null;
        document.getElementById('modalTitle').innerText = 'Add New Bank Entry';
        document.getElementById('submitEntryBtn').innerText = '+ Add';

        const valInputs = document.querySelectorAll('.val-input');
        valInputs.forEach(input => input.value = 0);

        setCurrentDateTime();
        const modal = document.getElementById('entryModal');
        modal.style.display = 'flex';

        const firstInput = modal.querySelector('.val-input');
        if (firstInput) {
            firstInput.focus();
            firstInput.select();
        }
    }

    function openEditModal(button) {
        const tr = button.closest('tr');
        editingRow = tr;

        document.getElementById('modalTitle').innerText = 'Edit Bank Entry';
        document.getElementById('submitEntryBtn').innerText = 'Update';

        const storedValues = JSON.parse(tr.getAttribute('data-values'));
        const storedDateTime = tr.getAttribute('data-datetime');

        const valInputs = document.querySelectorAll('.val-input');
        for (let i = 0; i < valInputs.length; i++) {
            valInputs[i].value = storedValues[i] ?? 0;
        }
        document.getElementById('entryDateTime').value = storedDateTime;

        const modal = document.getElementById('entryModal');
        modal.style.display = 'flex';

        const firstInput = modal.querySelector('.val-input');
        if (firstInput) {
            firstInput.focus();
            firstInput.select();
        }
    }

    // Open Glass Delete Confirm Modal
    function openDeleteModal(button) {
        rowToDelete = button.closest('tr');
        const dateText = rowToDelete.querySelector('.datetime-badge')?.innerText.replace('\n', ' ') || '';
        const sumText = rowToDelete.cells[3]?.innerText || '';

        document.getElementById('deletePreviewInfo').innerHTML =
            `<strong>Entry:</strong> ${dateText} &nbsp;|&nbsp; <strong>Total:</strong> ${sumText}`;

        document.getElementById('deleteConfirmModal').style.display = 'flex';
        document.getElementById('cancelDeleteBtn').focus();
    }

    function closeDeleteModal() {
        rowToDelete = null;
        document.getElementById('deleteConfirmModal').style.display = 'none';
    }

    function confirmDeleteRow() {
        if (rowToDelete) {
            rowToDelete.remove();
        }
        closeDeleteModal();
    }

    function outsideDeleteClick(event) {
        if (event.target === document.getElementById('deleteConfirmModal')) {
            closeDeleteModal();
        }
    }

    function closeModal() {
        editingRow = null;
        document.getElementById('entryModal').style.display = 'none';
    }

    function outsideClick(event) {
        if (event.target === document.getElementById('entryModal')) {
            closeModal();
        }
    }

    function syncDropdownsFromInput() {
        const dateTimeVal = document.getElementById('entryDateTime').value;
        if (dateTimeVal) {
            const dateObj = new Date(dateTimeVal);
            document.getElementById('monthSelect').value = dateObj.getMonth();
            document.getElementById('yearSelect').value = dateObj.getFullYear();
        }
    }

    function buildDateTimeCellHTML(dateTimeStr) {
        const dt = new Date(dateTimeStr);
        const day = String(dt.getDate()).padStart(2, '0');
        const month = String(dt.getMonth() + 1).padStart(2, '0');
        const year = dt.getFullYear();
        const time = dt.toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit', hour12: true });

        return `
            <div class="date-action-wrapper">
                <div class="row-actions">
                    <button type="button" class="btn-row-edit" onclick="openEditModal(this)">Edit</button>
                    <button type="button" class="btn-row-delete" onclick="openDeleteModal(this)">Delete</button>
                </div>
                <div class="datetime-badge">${day}-${month}-${year}<span class="time-sub">${time}</span></div>
            </div>
        `;
    }

    function loadLastAddedValues() {
        if (!lastAddedValues) {
            alert("No previous entry found!");
            return;
        }
        const valInputs = document.querySelectorAll('.val-input');
        for (let i = 0; i < valInputs.length; i++) {
            valInputs[i].value = lastAddedValues[i];
        }
        valInputs[0].focus();
        valInputs[0].select();
    }

    function populateRow(tr, entryDateTime, currentValues, monthBeginning) {
        const bankLabels = document.querySelectorAll('.bank-label');
        let sumOfValues = 0;
        let banksHTML = `<ul class="clean-list">`;
        let valuesHTML = `<ul class="clean-list val-list">`;

        for (let i = 0; i < 5; i++) {
            const bankName = bankLabels[i].innerText;
            const amount = currentValues[i];
            sumOfValues += amount;

            banksHTML += `<li><strong>${bankName}</strong></li>`;
            valuesHTML += `<li>₹${amount.toLocaleString('en-IN')}</li>`;
        }
        banksHTML += `</ul>`;
        valuesHTML += `</ul>`;

        const remainingBalance = monthBeginning - sumOfValues;

        tr.setAttribute('data-sum', sumOfValues);
        tr.setAttribute('data-values', JSON.stringify(currentValues));
        tr.setAttribute('data-datetime', entryDateTime);

        tr.innerHTML = `
            <td>${buildDateTimeCellHTML(entryDateTime)}</td>
            <td>${banksHTML}</td>
            <td>${valuesHTML}</td>
            <td class="highlight-val">₹${sumOfValues.toLocaleString('en-IN')}</td>
            <td class="remaining-val">${monthBeginning.toLocaleString('en-IN')} - ${sumOfValues.toLocaleString('en-IN')} = ₹${remainingBalance.toLocaleString('en-IN')}</td>
        `;
    }

    function saveEntry() {
        const monthBeginning = parseFloat(document.getElementById('monthBeginning').value) || 0;
        const entryDateTime = document.getElementById('entryDateTime').value;
        
        if (!entryDateTime) {
            alert("Please select a date and time!");
            return;
        }

        const valInputs = document.querySelectorAll('.val-input');
        let currentValues = [];
        for (let i = 0; i < 5; i++) {
            currentValues.push(parseFloat(valInputs[i].value) || 0);
        }

        lastAddedValues = currentValues;

        if (editingRow) {
            populateRow(editingRow, entryDateTime, currentValues, monthBeginning);
        } else {
            const tbody = document.getElementById('outputBody');
            const tr = document.createElement('tr');
            populateRow(tr, entryDateTime, currentValues, monthBeginning);
            tbody.prepend(tr);
        }

        valInputs.forEach(input => input.value = 0);
        closeModal();
    }

    function recalculateAll() {
        const monthBeginning = parseFloat(document.getElementById('monthBeginning').value) || 0;
        const rows = document.querySelectorAll('#outputBody tr');

        rows.forEach(row => {
            const sum = parseFloat(row.getAttribute('data-sum')) || 0;
            const remaining = monthBeginning - sum;
            row.cells[4].innerText = `${monthBeginning.toLocaleString('en-IN')} - ${sum.toLocaleString('en-IN')} = ₹${remaining.toLocaleString('en-IN')}`;
        });
    }
